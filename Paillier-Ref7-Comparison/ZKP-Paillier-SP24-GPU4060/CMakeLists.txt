cmake_minimum_required(VERSION 3.20)
project(ZKP-Paillier-SP24-GPU2 VERSION 2.0.0 LANGUAGES CXX)
option(ZKP_ENABLE_CUDA "Build the NVIDIA CUDA/CGBN backend" OFF)
option(ZKP_ENABLE_OPENMP "Enable CPU OpenMP workers" ON)
option(ZKP_BUILD_TESTS "Build correctness tests" ON)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_path(NTL_INCLUDE_DIR NTL/ZZ.h REQUIRED)
find_library(NTL_LIBRARY NAMES ntl REQUIRED)
find_path(GMP_INCLUDE_DIR gmp.h PATH_SUFFIXES x86_64-linux-gnu aarch64-linux-gnu REQUIRED)
find_library(GMP_LIBRARY NAMES gmp REQUIRED)
find_package(Threads REQUIRED)
include(cmake/CheckToolchain.cmake)
zkp_check_ntl_abi()
if(ZKP_ENABLE_OPENMP)
  find_package(OpenMP REQUIRED COMPONENTS CXX)
endif()
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS
  src/cpp/app/*.cpp src/cpp/app/math/*.cpp src/cpp/app/utils/*.cpp)
add_library(zkp_paillier STATIC ${CORE_SOURCES} src/cpp/app/acceleration/Backend.cpp)
target_include_directories(zkp_paillier PUBLIC src/cpp ${NTL_INCLUDE_DIR} ${GMP_INCLUDE_DIR})
target_link_libraries(zkp_paillier PUBLIC ${NTL_LIBRARY} ${GMP_LIBRARY} Threads::Threads)
if(ZKP_ENABLE_OPENMP)
  target_link_libraries(zkp_paillier PUBLIC OpenMP::OpenMP_CXX)
endif()
if(ZKP_ENABLE_CUDA)
  # RTX 4060 / RTX 4060 Laptop: native Ada SASS, no PTX JIT requirement.
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 89-real CACHE STRING "CUDA architectures")
  endif()
  zkp_select_cuda_host()
  enable_language(CUDA)
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 11.8)
    message(FATAL_ERROR "RTX 4060 needs CUDA >= 11.8. Ubuntu 26.04: use CUDA 13.4 and the system g++.")
  endif()
  find_package(CUDAToolkit REQUIRED)
  # Link the imported runtime target explicitly, including for C++ executables
  # that consume zkp_cuda. Avoid implicit -lcudart paths from a previous cache.
  set(CMAKE_CUDA_RUNTIME_LIBRARY None)
  add_library(zkp_cuda STATIC src/cpp/app/acceleration/CudaBackend.cu)
  set_target_properties(zkp_cuda PROPERTIES CUDA_STANDARD 14 CUDA_STANDARD_REQUIRED ON)
  target_include_directories(zkp_cuda PRIVATE third_party/cgbn/include ${GMP_INCLUDE_DIR})
  target_compile_options(zkp_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--expt-relaxed-constexpr>)
  # Static CUDA runtime avoids dependence on a shell-specific LD_LIBRARY_PATH.
  target_link_libraries(zkp_cuda PUBLIC CUDA::cudart_static ${GMP_LIBRARY})
  target_link_libraries(zkp_paillier PUBLIC zkp_cuda)
  target_compile_definitions(zkp_paillier PUBLIC ZKP_HAS_CUDA=1)
  add_executable(zkp_device_info tools/device_info.cu)
  set_target_properties(zkp_device_info PROPERTIES CUDA_STANDARD 14 CUDA_STANDARD_REQUIRED ON)
  target_link_libraries(zkp_device_info PRIVATE CUDA::cudart_static)
else()
  target_sources(zkp_paillier PRIVATE src/cpp/app/acceleration/CudaStub.cpp)
endif()
add_executable(zkp_bench tools/benchmark.cpp)
target_link_libraries(zkp_bench PRIVATE zkp_paillier)
configure_file(cmake/build-info.txt.in build-info.txt @ONLY)
message(STATUS "ZKP host compiler: ${CMAKE_CXX_COMPILER} (${CMAKE_CXX_COMPILER_VERSION})")
message(STATUS "ZKP NTL: ${NTL_LIBRARY}; GMP: ${GMP_LIBRARY}")
if(ZKP_ENABLE_CUDA)
  message(STATUS "ZKP CUDA: ${CMAKE_CUDA_COMPILER_VERSION}; host: ${CMAKE_CUDA_HOST_COMPILER}; arch: ${CMAKE_CUDA_ARCHITECTURES}")
endif()
if(ZKP_BUILD_TESTS)
  enable_testing()
  add_executable(acceleration_tests tests/acceleration_tests.cpp)
  target_link_libraries(acceleration_tests PRIVATE zkp_paillier)
  add_executable(transcript_fixture tests/v6_transcript_fixture.cpp)
  target_link_libraries(transcript_fixture PRIVATE zkp_paillier)
  add_test(NAME cpu_correctness COMMAND acceleration_tests)
  add_test(NAME cuda_correctness COMMAND acceleration_tests --cuda)
  set_tests_properties(cuda_correctness PROPERTIES SKIP_RETURN_CODE 77 LABELS "gpu")
  add_test(NAME cpu_protocol_smoke COMMAND zkp_bench --backend cpu --bits 64
           --messages 3 --range-proofs 3 --threads 2 --repeat 1)
  set_tests_properties(cpu_protocol_smoke PROPERTIES TIMEOUT 120)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME gpu_acceptance_guards COMMAND ${Python3_EXECUTABLE}
             ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_gpu_acceptance.py)
  endif()
endif()

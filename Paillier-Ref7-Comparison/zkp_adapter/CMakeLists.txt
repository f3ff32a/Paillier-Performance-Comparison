cmake_minimum_required(VERSION 3.20)
project(PaillierRef7ZkpAdapter LANGUAGES CXX)
option(ZKP_COMPARE_CUDA "Build the original ZKP CUDA backend" OFF)
set(ZKP_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../ZKP-Paillier-SP24-GPU4060" CACHE PATH
    "Root of the supplied ZKP-Paillier-SP24-GPU4060 sources")
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
foreach(required IN ITEMS src/cpp/app/acceleration/Backend.cpp
    src/cpp/app/acceleration/CudaStub.cpp src/cpp/app/acceleration/CudaBackend.cu
    src/cpp/vendor/json/json.hpp)
  if(NOT EXISTS "${ZKP_SOURCE_DIR}/${required}")
    message(FATAL_ERROR "Missing ${ZKP_SOURCE_DIR}/${required}; set ZKP_SOURCE_DIR")
  endif()
endforeach()
find_path(GMP_INCLUDE_DIR NAMES gmp.h PATH_SUFFIXES x86_64-linux-gnu aarch64-linux-gnu REQUIRED)
find_library(GMP_LIBRARY NAMES gmp REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED COMPONENTS CXX)
add_library(compare_backend STATIC "${ZKP_SOURCE_DIR}/src/cpp/app/acceleration/Backend.cpp")
target_include_directories(compare_backend PUBLIC "${ZKP_SOURCE_DIR}/src/cpp" "${GMP_INCLUDE_DIR}")
target_link_libraries(compare_backend PUBLIC "${GMP_LIBRARY}" Threads::Threads OpenMP::OpenMP_CXX)
if(ZKP_COMPARE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 89-real CACHE STRING "CUDA GPU architecture")
  endif()
  # CPU/CUDA host compiler must use the same ABI; callers may set both explicitly.
  if(NOT DEFINED CMAKE_CUDA_HOST_COMPILER)
    set(CMAKE_CUDA_HOST_COMPILER "${CMAKE_CXX_COMPILER}" CACHE FILEPATH "CUDA host compiler")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CUDA_RUNTIME_LIBRARY None)
  add_library(compare_cuda STATIC "${ZKP_SOURCE_DIR}/src/cpp/app/acceleration/CudaBackend.cu")
  set_target_properties(compare_cuda PROPERTIES CUDA_STANDARD 14 CUDA_STANDARD_REQUIRED ON)
  target_include_directories(compare_cuda PRIVATE "${ZKP_SOURCE_DIR}/third_party/cgbn/include" "${GMP_INCLUDE_DIR}")
  target_compile_options(compare_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--expt-relaxed-constexpr>)
  target_link_libraries(compare_cuda PUBLIC CUDA::cudart_static "${GMP_LIBRARY}")
  target_link_libraries(compare_backend PUBLIC compare_cuda)
  target_compile_definitions(compare_backend PUBLIC ZKP_HAS_CUDA=1)
else()
  target_sources(compare_backend PRIVATE "${ZKP_SOURCE_DIR}/src/cpp/app/acceleration/CudaStub.cpp")
endif()
add_executable(paillier_compare paillier_compare.cpp)
target_link_libraries(paillier_compare PRIVATE compare_backend)
find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  enable_testing()
  add_test(NAME adapter_contract COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/test_adapter.py" --binary "$<TARGET_FILE:paillier_compare>")
endif()

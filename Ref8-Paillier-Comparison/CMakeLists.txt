cmake_minimum_required(VERSION 3.20)
project(Ref8PaillierComparison LANGUAGES CXX)
option(COMPARE_ENABLE_CUDA "Build both real CUDA backends" ON)
set(COMPARE_BITS "512;1024;2048" CACHE STRING "Requested public modulus sizes for ref8 binaries")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_path(GMP_INCLUDE_DIR gmp.h PATH_SUFFIXES x86_64-linux-gnu aarch64-linux-gnu REQUIRED)
find_library(GMP_LIBRARY NAMES gmp REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED COMPONENTS CXX)
set(USER_ACCEL "${CMAKE_CURRENT_SOURCE_DIR}/vendor/user_acceleration")
add_library(user_accel STATIC "${USER_ACCEL}/Backend.cpp")
target_include_directories(user_accel PUBLIC "${USER_ACCEL}" "${GMP_INCLUDE_DIR}")
target_link_libraries(user_accel PUBLIC "${GMP_LIBRARY}" Threads::Threads OpenMP::OpenMP_CXX)
if(COMPARE_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 89-real CACHE STRING "RTX 4060 native architecture")
  endif()
  # Supply the same host C++ compiler to nvcc and the C++ adapter.
  if(NOT DEFINED CMAKE_CUDA_HOST_COMPILER)
    set(CMAKE_CUDA_HOST_COMPILER "${CMAKE_CXX_COMPILER}" CACHE FILEPATH "CUDA host compiler")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CUDA_RUNTIME_LIBRARY None)
  add_library(user_cuda STATIC "${USER_ACCEL}/CudaBackend.cu")
  target_include_directories(user_cuda PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/cgbn/include" "${GMP_INCLUDE_DIR}")
  set_target_properties(user_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(user_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--expt-relaxed-constexpr>)
  target_link_libraries(user_cuda PUBLIC CUDA::cudart_static "${GMP_LIBRARY}")
  target_link_libraries(user_accel PUBLIC user_cuda)
  target_compile_definitions(user_accel PUBLIC ZKP_HAS_CUDA=1)
  foreach(bits IN LISTS COMPARE_BITS)
    if(NOT bits MATCHES "^(512|1024|2048|3072|4096)$")
      message(FATAL_ERROR "COMPARE_BITS accepts 512,1024,2048,3072,4096")
    endif()
    math(EXPR prime_bits "${bits} / 2")
    add_executable(ref8_bench_${bits} adapters/ref8_bench.cu)
    target_include_directories(ref8_bench_${bits} PRIVATE "${GMP_INCLUDE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/cgbn/include")
    target_compile_definitions(ref8_bench_${bits} PRIVATE KEY_BITS=${prime_bits} CGBN_TPI=32)
    target_compile_options(ref8_bench_${bits} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--expt-relaxed-constexpr>)
    set_target_properties(ref8_bench_${bits} PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_link_libraries(ref8_bench_${bits} PRIVATE CUDA::cudart_static "${GMP_LIBRARY}")
  endforeach()
else()
  target_sources(user_accel PRIVATE "${USER_ACCEL}/CudaStub.cpp")
endif()
add_executable(user_bench adapters/user_bench.cpp)
target_link_libraries(user_bench PRIVATE user_accel)
